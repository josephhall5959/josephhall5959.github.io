\begin{frame}{Economics on One Slide}

\begin{itemize}
\item There is an island with 2 kinds of people:
\begin{itemize}
\item Banana farmers can only produce bananas
\item Apple farmers can only produce apples
\end{itemize}
\item Every farmer would prefer to have both apples and bananas \pause
\item Solution: a \textbf{market} emerges for trading bananas for apples \pause
\item In market equilibrium, farmers can trade $p$ bananas for one apple
\item Everyone is happier because they can eat both bananas and apples
\end{itemize}

\end{frame}

\begin{frame}
\frametitle{Finance in Two Slides (1): Time}
\bi
\m There is an island with 2 kinds of people:
\bi
\m Gatherers can find food in spring, but not in fall
\m Farmers can plant crops in spring, but can only harvest in fall
\ei
\pause
\m Solution: a \textbf{market} emerges for trading food \textbf{over time} \pause
\m Gatherers feed farmers in spring; farmers promise to pay it back in fall
\m In equilibrium, gatherers pay $p$ units of spring food, for a promise of one unit of fall food \pause
\m Everyone is happier because they get to eat in spring and fall
\ei
\end{frame}

\begin{frame}
\frametitle{Finance in Two Slides (2): Risk}
\bi
\m There is an island with 2 kinds of people:
\bi
\m Hunters can hunt meat, but they have 1/2 chance of finding nothing
\m Farmers can plant crops, which is safe, but meat tastes better
\ei \pause
\m Solution: a market emerges for trading \textbf{risky food for safe food} \pause
\m Farmers form a venture-capital group which pays the hunter.io startup to hunt meat \pause
\m Everyone is happier: if hunters succeed, everyone gets meat. If they fail, hunters don't starve
\ei
\end{frame}

\begin{frame}
\frametitle{Markets for Time and Risk}
\bi
\m Financial markets allow people to trade \textbf{time} and \textbf{risk}
\m This allows people to consume time/risk bundles different from what they started with
\ei
\end{frame}

\begin{frame}
\frametitle{Markets for Time}
\bi
\m Companies need to pay up front for equipment/research paying off over many years \pause (Stocks, bonds, loans) \pause
\m An online shop needs to pay up front to buy and warehouse inventory, then sell it over a few months \pause (Trade credit) \pause
\m You are poor now, but if you do well in this class, you might be rich after you graduate \pause (Student loans) \pause
\m When you buy things like houses \& cars, you need to pay a lot up front, and you enjoy the benefits over time \pause (Mortgages, auto loans) \pause
\ei
\end{frame}

\begin{frame}
\frametitle{Markets for Risk}
\bi 
\m A tech startup needs to invest in an idea that might succeed or fail \pause (Stocks, bonds, venture capital) \pause
\m Shell drills an oil well, but doesn't know whether Russia and Saudi Arabia will start an oil price war \pause (Oil futures/forwards) \pause
\m If you crash your car, you will need to spend a lot of money \pause (Car insurance) \pause
\m If you get sick, you will need to spend even more money \pause (Health insurance)
\ei
\end{frame}

\begin{frame}
\frametitle{Real Assets}

Time and risk are traded using \textbf{financial assets}\pause

\vspace{2ex}

\textbf{Real assets}
\begin{itemize}
\item Real things that produce real goods and services 
\item Factories, machines, software, vaccine designs\ldots
\end{itemize} 

\end{frame}

\begin{frame}
\frametitle{Trading Promises}

Time and risk are traded using \textbf{financial assets}\pause

\vspace{2ex}

\textbf{Financial assets}
\begin{itemize}
\item Wikipedia: ``A financial asset is a non-physical asset whose value is derived from a contractual claim, such as bank deposits, bonds, and stocks''
\m Financial assets are \textbf{promises} to deliver something (money, goods) at some point \pause
\m Different kinds of promises allow people to trade different kinds of time/risk:
\bi \pause
\m Fixed \$ amount (bond), based on company performance (stock), or based on other numbers (derivatives) \pause
\m ``Secured'' by real production of real assets (secured bonds, mortgages) or not \pause
\m Sooner or later (bond term), riskier or safer \pause
\ei
\end{itemize}

\vspace{2ex}

See \href{https://anthonyleezhang.substack.com/p/the-market-for-promises}{The Market for Promises}

\end{frame}
